## 6. 通行图和距离更新

Policy `.cpp` 内部只建一次整栋图：普通格节点、结构 `WalkableCells` 节点、普通格互相开口边、结构 `InternalConnections` 和结构 `Openings`。地址按 Z/Y/X 稳定排序。

最小实现形态：

```cpp
struct FTraversalGraph
{
	TArray<FIntVector> Addresses;
	TMap<FIntVector, int32> NodeByAddress;
	TArray<TArray<int32>> Neighbors;
};

void UpdateNearestDistances(
	const FTraversalGraph& Graph,
	const TConstArrayView<int32> SourceNodes,
	TArray<int32>& InOutNearestDistances)
{
	TQueue<int32> Queue;
	for (const int32 SourceNode : SourceNodes)
	{
		if (InOutNearestDistances.IsValidIndex(SourceNode)
			&& InOutNearestDistances[SourceNode] > 0)
		{
			InOutNearestDistances[SourceNode] = 0;
			Queue.Enqueue(SourceNode);
		}
	}

	int32 Node = INDEX_NONE;
	while (Queue.Dequeue(Node))
	{
		const int32 NextDistance = InOutNearestDistances[Node] + 1;
		for (const int32 Neighbor : Graph.Neighbors[Node])
		{
			if (NextDistance < InOutNearestDistances[Neighbor])
			{
				InOutNearestDistances[Neighbor] = NextDistance;
				Queue.Enqueue(Neighbor);
			}
		}
	}
}
```

地图只有约 72～160 个可走节点。每接受一处内容做一次多源距离更新比全点对矩阵、回溯搜索或运行时对象缓存更直接。

## 7. 候选结构：一个位置可以有多个朝向，但不能因此被重复加权

```cpp
struct FPlacementVariant
{
	TArray<FTransform> LocalSpawnTransforms;
	TArray<FIntVector> ResourceBlockedAddresses;
};

struct FPlacementCandidate
{
	EPopulationPlacementKind Kind = EPopulationPlacementKind::SpikeTrap;
	FIntVector AnchorAddress = FIntVector::ZeroValue;
	TArray<int32> SpacingNodeIndices;
	TArray<FPlacementVariant> Variants;
};
```

- 地刺：一个格是一个候选；每个真实开口生成一个“正面进入方向”变体。
- 冲锤：一个格是一个候选；每个封闭墙面生成一个变体。
- 发射器：一个格是一个候选；每个“前方开放且连接普通格、后方封闭”的方向生成一个变体。
- 先抽中位置，再在 `Variants` 中均匀抽一个朝向，因此转角有两种合法安装方向，不会在类型权重中算成两个位置。

方向和墙面变换核心代码：

```cpp
FVector DirectionVector(const uint8 Direction)
{
	switch (Direction)
	{
	case 0: return FVector(0.0, 1.0, 0.0);   // North
	case 1: return FVector(1.0, 0.0, 0.0);   // East
	case 2: return FVector(0.0, -1.0, 0.0);  // South
	default: return FVector(-1.0, 0.0, 0.0); // West
	}
}

FTransform MakeWallMountedTransform(
	const FVector& CellFloorCenter,
	const FVector& ForwardIntoCell,
	const double HalfTileCm,
	const double WallInsetCm,
	const double MountHeightCm)
{
	const FVector Rearward = -ForwardIntoCell;
	const FVector Location = CellFloorCenter
		+ Rearward * (HalfTileCm - WallInsetCm)
		+ FVector(0.0, 0.0, MountHeightCm);
	return FTransform(ForwardIntoCell.Rotation(), Location);
}
```

高厅摆锤不参加普通机关权重抽取：

```cpp
bool BuildMandatoryPendulum(
	const FZeroEscapeGeneratedLevelPlan& LevelPlan,
	const FZeroEscapeGeneratedStructure& Structure,
	const double FloorTopZCm,
	FPlacementCandidate& OutCandidate)
{
	if (Structure.Kind != EZeroEscapeStructureKind::HighCeilingRoom
		|| Structure.WalkableCells.Num() != 2
		|| Structure.InternalConnections.Num() != 1)
	{
		return false;
	}

	const FIntVector First = Structure.WalkableCells[0];
	const FIntVector Second = Structure.WalkableCells[1];
	if (First.Z != Second.Z
		|| FMath::Abs(First.X - Second.X) + FMath::Abs(First.Y - Second.Y) != 1)
	{
		return false;
	}

	const FVector FirstFloor = AddressFloorLocation(LevelPlan, First, FloorTopZCm);
	const FVector SecondFloor = AddressFloorLocation(LevelPlan, Second, FloorTopZCm);
	FPlacementVariant Variant;
	Variant.LocalSpawnTransforms.Add(FTransform(
		FRotator(0.0, Structure.QuarterTurnCount * -90.0, 0.0),
		(FirstFloor + SecondFloor) * 0.5));
	Variant.ResourceBlockedAddresses = { First, Second };

	OutCandidate.Kind = EPopulationPlacementKind::Pendulum;
	OutCandidate.AnchorAddress = Structure.BaseCoordinate;
	OutCandidate.Variants = { MoveTemp(Variant) };
	return true;
}
```

## 8. 机关阶段：类型先抽、位置后抽

```cpp
while (RemainingOrdinaryHazards > 0)
{
	TArray<FFeasibleHazardPool> FeasiblePools;
	BuildFeasiblePool(
		EPopulationPlacementKind::SpikeTrap,
		DifficultySettings.Hazards.SpikeTrapWeight,
		SpikeCandidates,
		NearestHazardDistance,
		DifficultySettings.Hazards.MinimumRouteSpacingTiles,
		FeasiblePools);
	BuildFeasiblePool(
		EPopulationPlacementKind::BatteringRam,
		DifficultySettings.Hazards.BatteringRamWeight,
		RamCandidates,
		NearestHazardDistance,
		DifficultySettings.Hazards.MinimumRouteSpacingTiles,
		FeasiblePools);
	BuildFeasiblePool(
		EPopulationPlacementKind::GuidedLauncher,
		DifficultySettings.Hazards.GuidedLauncherWeight,
		LauncherCandidates,
		NearestHazardDistance,
		DifficultySettings.Hazards.MinimumRouteSpacingTiles,
		FeasiblePools);

	if (FeasiblePools.IsEmpty())
	{
		WorkingPlan.HazardStats.CandidateExhaustedCount =
			RemainingOrdinaryHazards;
		break;
	}

	const int32 PoolIndex = PickWeightedPool(FeasiblePools, HazardRng);
	const FFeasibleHazardPool& Pool = FeasiblePools[PoolIndex];
	const int32 CandidateIndex =
		Pool.CandidateIndices[HazardRng.RandRange(0, Pool.CandidateIndices.Num() - 1)];
	const FPlacementCandidate& Candidate = Pool.SourceCandidates[CandidateIndex];
	const FPlacementVariant& Variant =
		Candidate.Variants[HazardRng.RandRange(0, Candidate.Variants.Num() - 1)];

	AcceptPlacement(Candidate, Variant, WorkingPlan.HazardPlacements);
	UpdateNearestDistances(Graph, Candidate.SpacingNodeIndices, NearestHazardDistance);
	--RemainingOrdinaryHazards;
}
```

`BuildFeasiblePool` 只把“权重大于 0 且当前至少有一个满足间距的位置”的类型放入池。某类合法位置耗尽后自然退出本轮，不转移成固定保底，也不缩小间距。

## 9. 资源阶段：先确定所有格，再生成格内随机位置

```cpp

while (SelectedResourceAnchors.Num() < ResourceTarget
	&& !RemainingResourceAnchors.IsEmpty())

		0, RemainingResourceAnchors.Num() - 1);
	const FIntVector Address = RemainingResourceAnchors[PickIndex];

		TConstArrayView<int32>(&SourceNode, 1),

// 宏观锚点已经全部固定；从此处开始的随机数不会改变资源所在格。
const double SafeHalfRangeCm = LevelPlan.LogicalTileSizeCm * 0.5
	- ResourceAssembly.PlacementFootprintRadiusCm;

for (const FIntVector Address : SelectedResourceAnchors)
{
	const FVector CellFloorCenter =
		AddressFloorLocation(LevelPlan, Address, FloorTopZCm);
	const float SafeHalfRange = static_cast<float>(SafeHalfRangeCm);
	const FVector Offset(
		ResourceRng.FRandRange(-SafeHalfRange, SafeHalfRange),
		ResourceRng.FRandRange(-SafeHalfRange, SafeHalfRange),
		ResourceAssembly.SpawnZOffsetCm);

	FPopulationPlannedPlacement Placement;
	Placement.Kind = EPopulationPlacementKind::MagneticResource;
	Placement.AnchorAddress = Address;
	Placement.LocalSpawnTransforms.Add(FTransform(
		FQuat::Identity,
```

```cpp

&& PlacementFootprintRadiusCm < LogicalTileSizeCm * 0.5f
```

资源目标未达到时，Policy 仍返回 `Success`，但写入 `ResourceStats.ActualCount < TargetCount`；Populator 输出一条 Warning。Class 加载或 Spawn 失败仍是事务失败。

## 10. BuildPlan 入口和原子输出

```cpp
EPopulationPlacementResult FPopulationPlacementPolicy::BuildPlan(
	const FZeroEscapeGeneratedLevelPlan& LevelPlan,
	const double FloorTopZCm,
	const FZeroEscapeHazardPopulationAssembly& HazardAssembly,
	const FZeroEscapeResourcePopulationAssembly& ResourceAssembly,
	const FZeroEscapePopulationDifficultySettings& DifficultySettings,
	FPopulationPlacementPlan& OutPlan,
	FString& OutError)
{
	OutPlan = FPopulationPlacementPlan();
	OutError.Reset();
	FPopulationPlacementPlan WorkingPlan;

	if (!ValidateInputs(
			LevelPlan,
			FloorTopZCm,
			HazardAssembly,
			ResourceAssembly,
			DifficultySettings,
			OutError))
	{
		return EPopulationPlacementResult::InvalidConfiguration;
	}

	FTraversalGraph Graph;
	if (!BuildTraversalGraph(LevelPlan, Graph, OutError))
	{
		return EPopulationPlacementResult::InvalidTraversalGraph;
	}

	const int32 GameplayAreaCellCount = CountGameplayAreaCells(LevelPlan);
	WorkingPlan.HazardStats.TargetCount = FMath::Max(
		RoundDensityTarget(
			GameplayAreaCellCount,
			DifficultySettings.Hazards.ExpectedHazardsPer100GameplayCells),
		CountLegalHighCeilingRooms(LevelPlan));
	WorkingPlan.ResourceStats.TargetCount = RoundDensityTarget(
		GameplayAreaCellCount,
		DifficultySettings.Resources.ExpectedResourcesPer100GameplayCells);

	FRandomStream HazardRng = FGenerationCore::MakeRandomStream(
		LevelPlan.Signature.Seed,
		GAlgorithmVersion,
		ERandomDomain::HazardPopulationPlacement);
	FRandomStream ResourceRng = FGenerationCore::MakeRandomStream(
		LevelPlan.Signature.Seed,
		GAlgorithmVersion,
		ERandomDomain::ResourcePopulationPlacement);

	if (!PlanHazards(/*...*/, WorkingPlan, OutError)
		|| !PlanResources(/*...*/, WorkingPlan, OutError))
	{
		return EPopulationPlacementResult::InvalidPlan;
	}

	if (!ValidateActorBudget(WorkingPlan, OutError))
	{
		return EPopulationPlacementResult::SpawnBudgetExceeded;
	}

	OutPlan = MoveTemp(WorkingPlan);
	return EPopulationPlacementResult::Success;
}
```

正式实现会让 `PlanHazards` / `PlanResources` 返回具体枚举；上面的省略形态只表达原子输出顺序，不把规划失败混进 Spawn 预算错误。

## 11. GameplayPopulator 接入

`AZeroEscapeGameplayPopulator` 的公开 API 和 `SpawnedActors` 不变。`Populate` 改为一次规划、一次验证、一次 Spawn：

```cpp
bool AZeroEscapeGameplayPopulator::Populate(
	AZeroEscapeRuntimeLevelGenerator& Generator)
{
	ClearPopulation();
	UWorld* World = GetWorld();
	if (!IsValid(World) || !IsValid(PopulationProfile))
	{
		return FailPopulation(TEXT("InvalidSetup"));
	}

	FZeroEscapeGeneratedLevelPlan LevelPlan;
	FTransform GeneratedRootWorldTransform;
	double FloorTopZCm = 0.0;
	if (!Generator.GetGeneratedPopulationSnapshot(
			LevelPlan, GeneratedRootWorldTransform, FloorTopZCm))
	{
		return FailPopulation(TEXT("PopulationSnapshotFailed"));
	}

	const FZeroEscapePopulationDifficultySettings* Difficulty =
		FindUniqueDifficultySettings(
			*PopulationProfile, LevelPlan.Signature.Difficulty);
	if (Difficulty == nullptr)
	{
		return FailPopulation(TEXT("DifficultySettingsInvalid"));
	}

	FPopulationPlacementPlan PlacementPlan;
	FString PlacementError;
	const EPopulationPlacementResult Result =
		FPopulationPlacementPolicy::BuildPlan(
			LevelPlan,
			FloorTopZCm,
			PopulationProfile->HazardAssembly,
			PopulationProfile->ResourceAssembly,
			*Difficulty,
			PlacementPlan,
			PlacementError);
	if (Result != EPopulationPlacementResult::Success)
	{
		return FailPopulation(Result, PlacementError);
	}

	TMap<EPopulationPlacementKind, UClass*> LoadedClasses;
	if (!LoadOnlyUsedClasses(
			PlacementPlan,
			PopulationProfile->HazardAssembly,
			PopulationProfile->ResourceAssembly,
			LoadedClasses))
	{
		return FailPopulation(TEXT("ActorClassLoadFailed"));
	}

	TArray<FTransform> WorldTransforms;
	if (!BuildAndValidateWorldTransforms(
			PlacementPlan,
			GeneratedRootWorldTransform,
			WorldTransforms))
	{
		return FailPopulation(TEXT("WorldTransformInvalid"));
	}

	FActorSpawnParameters SpawnParameters;
	SpawnParameters.Owner = this;
	SpawnParameters.SpawnCollisionHandlingOverride =
		ESpawnActorCollisionHandlingMethod::AlwaysSpawn;

	for (const FPopulationPlannedPlacement& Placement :
		IterateHazardsThenResources(PlacementPlan))
	{
		UClass* ActorClass = LoadedClasses.FindRef(Placement.Kind);
		for (const FTransform& LocalTransform : Placement.LocalSpawnTransforms)
		{
			const FTransform WorldTransform =
				LocalTransform * GeneratedRootWorldTransform;
			AActor* Spawned = World->SpawnActor<AActor>(
				ActorClass, WorldTransform, SpawnParameters);
			if (!IsValid(Spawned))
			{
				return FailPopulation(TEXT("ActorSpawnFailed"));
			}
			SpawnedActors.Add(Spawned);
		}
	}

	LogPopulationSummary(LevelPlan, PlacementPlan);
	return true;
}
```

正式代码不会创建 `IterateHazardsThenResources` 临时容器；会用两个小循环避免复制。失败入口继续统一调用 `ClearPopulation()`，发射器拥有的预装弹体随 Launcher `EndPlay` 一起清理。

## 13. 明确删除和保留

删除：

- `FZeroEscapePlacementRule`、`Rules`；
- `OneEveryNCells`、`MaxCount`、`bStraightCorridorOnly`、`bAvoidStartExitNeighbors`；
- 旧 `EPopulationPlacementBudgetResult` 和 header-only `Evaluate`；
- 逐 Rule 候选查询、Shuffle、同步加载、立即 Spawn 链；
- 旧普通格候选 UFUNCTION。

保留：

- 一个 `AZeroEscapeGameplayPopulator`；
- 一个 `UZeroEscapePopulationProfile`；
- `SpawnedActors`、`AlwaysSpawn` 和原子回滚；
- `GetGeneratedSeed()`；
- 现有测试文件和 Public/Private 目录职责。

## 14. 实施时必须及时回报的情况

以下任何一项出现时暂停对应检查点并询问用户，不自行加兼容或降级：

1. 正式 DataAsset 中三档初值无法通过 288 Seed 资源满额目标。
2. 冲锤 450 cm 行程或发射器预装弹体在合法候选中仍穿结构。
3. 摆锤真实 79° 配置在 1×2 高厅发生墙体碰撞。
4. 随机磁力物导致导航持续重建、追猎者路径失效或明显堵路。
5. 删除旧查询前发现新的 Blueprint/资产调用。
6. 基线门禁时 Level0、动画或其他任务文件仍未确认归属。
